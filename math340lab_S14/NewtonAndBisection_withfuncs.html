% Starting interval:
a = 1;   b = 4;
Error = 1E-6;

% call function 'bisection'
[app_root, iterates] = bisection(f,a,b,Error,50);
fprintf('The root of the function is %10.6f\n',app_root)

% upper bound for error in bisection method is the length of the interval:
app_error = (b-a)./2.^(1:length(iterates));

% plot error vs iteration number
plot((1:length(iterates)),app_error)
xlabel('n'), ylabel('E_n')
title('Bisection method error')

% app_root is the approximation to the root found by the algorithm,
% iterates is a vector containing the iterates.
function  [app_root, iterates]  = bisection(f, a, b, error, iter_max)

    % initialize the number of iterations
    num_iter = 1;

    % test the interval length and number ot iterations for bisection:
    while ((b-a) >= error)

        % find mid point of the interval:
        c=(a+b)/2;

        % save iterates in a vector:
        iterates(num_iter) = c;

        % check the the signs of the endpoints of subintervals:
        if (sign(f(b))*sign(f(c))<=0)
            a=c;
        else
            b=c;
        end

        % count number of iterates:
        num_iter = num_iter + 1;

        if num_iter==iter_max
            printf('Maximum number of iterates (%d) reached, try bigger iter_max. app_root set to inf.', iter_max)
            app_root = inf;
            break;
        end

    end

    % root:
    app_root=c;
end

% app_root is the approximation to the root found by the algorithm,
% iterates is a vector containing the iterates.
function [app_root, iterates] = newton(f, df, x0, error, iter_max)
    % to evaluate the condition abs(xn_1 - xn) for the first time
    % in the while loop these two lines need to be present prior
    % to entering the code
    xn = x0;
    xn_1 = xn - f(xn) / df(xn);

    % save the first iterates x0 and x1.
    iterates(1) = xn; iterates(2) = xn_1;

    % iter keeps count of the number of iterations (2 thus far)
    iter = 2;

    % we iterate as long as the error between consecutive iterates is
    % greater than error
    while(abs(xn_1 - xn) >= error)
        if(iter > iter_max)
            fprintf('Maximum number of iterates (%d) reached, try bigger iter_max. app_root set to inf.', iter_max);
            app_root = inf;
            break;
        end
        % compute next iterate
        xn = xn_1;
        xn_1 = xn - f(xn) / df(xn);
        % save it
        iterates(iter) = xn_1;
        % increment iter
        iter = iter + 1;
    end
    app_root = xn_1;
end
