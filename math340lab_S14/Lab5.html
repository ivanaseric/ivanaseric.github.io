2, 3. (Solution)

Plot the partial sums for Taylor series for N = 1,3,5,7,9, on domain on the same plot. Make sure you it is clear from the plot which graph belongs to which N. Also, plot MATLAB function erf(x). 3. Plot the error for each N.

clear all
clc
% define vector x for plots
x = linspace(-2,2,1000);
Markers = ['o' 's' '^' 'x'];


figure(1)
hold on
figure(2)
hold on
for iter = 1:4
    [erfTaylorSum, PtError] = erfTaylor(x,iter);
    figure(1) % plot for S_n
    line_fewer_markers(x,erfTaylorSum,10-(iter),Markers((iter)),'MarkerSize',5);
    figure(2) % plot for the error $|S_n - erf(x) |$
    line_fewer_markers(x,abs( PtError),10-(iter),Markers((iter)),'MarkerSize',5);
    LegendNames{iter} = [ 'N = ',num2str(2*iter-1,'%i') ];
end
figure(2)
legend(LegendNames)
legend('Location','north')
xlabel('x'), ylabel('E_N')
title('Taylor series error for erf(x)')
box on

% plot the exact function
figure(1)
plot(x,erf(x),'LineWidth',2)
LegendNames{iter+1} = 'erf(x)';
legend(LegendNames)
legend('Location','southEast')
xlabel('x'), ylabel('S_N')
title('Partial sums of Taylor series for erf(x)')
box on
