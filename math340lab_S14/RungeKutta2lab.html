Use Runge Kutta second order method (write your own code as a MATLAB function) to solve the HW problems:

#1

Plot your solutions for with on the same axes with the exact solution. Calculate the order of convergence.

clear all; clc
f = @(x,y) (cos(y)).^2;
h = 2.^(-(0:6));
Markers = ['o' '^' 's' '>'  'd' 'p' '<'];
ErrorRK = zeros(1,length(h));
figure(1)
hold on
for i = 1: length(h)
    [x, y ] = RungeKutta2method(f, 0, 0, h(i), 10 );
    line_fewer_markers(x,y, 2*(10-i),Markers(i),'MarkerFaceColor','none','MarkerSize',5);
    % find error at the last point
    ErrorRK(i) = abs( y(end) - atan(x(end)));
end
plot(x, atan(x),'LineWidth',2)
legend('h = 1', 'h = 1/2', 'h = 1/4', 'h = 1/8', 'h = 1/16','h = 1/32','h = 1/64','exact')
legend('Location', 'SE')
xlabel('x'), ylabel('y')
box on

Order of convergence

p1 = polyfit(log(h),log(ErrorRK),1);
fprintf('Order of convergence is %3.2f \n',p1(1))

Order of convergence is 2.13

#2

Plot your solutions for with on the same axes with the exact solution. Comment on your results!

clear all; clc
f = @(x,y) 2.*sqrt(y)  -  2./x - 2./x.^3;
y_exact = @(x) (x + 1./x).^2;
h = 2.^(-(0:6));
Markers = ['o' '^' 's' '>'  'd' 'p' '<'];
figure(2)
hold on
for i = 1: length(h)
    [x, y ] = RungeKutta2method(f, 1, 4, h(i), 4 );
    line_fewer_markers(x,y, 2*(10-i),Markers(i),'MarkerFaceColor','none','MarkerSize',5);
    % find error at the last point
    ErrorRK(i) = abs( y(end) - y_exact(x(end)));
end
plot(x, y_exact(x),'LineWidth',2)
legend('h = 1', 'h = 1/2', 'h = 1/4', 'h = 1/8', 'h = 1/16','h = 1/32','h = 1/64','exact')
legend('Location', 'NW')
xlabel('x'), ylabel('y')
box on

Order of convergence

p1 = polyfit(log(h),log(ErrorRK),1);
fprintf('Order of convergence is %3.2f \n',p1(1))
