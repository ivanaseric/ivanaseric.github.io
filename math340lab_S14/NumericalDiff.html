Derivatives

clear all;  clc

f = @(x) exp(2.*x);
x = 0;      n = [1:10];     h = 1./2.^n;
dy(1,:) = (f(x+h) - f(x))./h;
dy(2,:) = (f(x+h)-2.*f(x) + f(x-h))./(h.^2);
dy(3,:) = (f(x+h) - f(x-h))./(2.*h);
dy(4,:) = (f(x+2.*h)-2.*f(x+h)+f(x))./(h.^2);

First derivative methods

figure(1)
line_fewer_markers(n, dy(1,:), 10, 'd', 'MarkerSize', 7);

title('First derivative')
xlabel('n'), ylabel('df/{dx}')
