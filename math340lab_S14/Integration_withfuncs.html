midpoint
function int_value = midpoint(f,a,b,n)
    h = (b-a)/n;
    S = 0;
    x = a + 0.5.*h;
    for i = 1:n
        S = S + f(x);
        x = x + h;
    end
    int_value = h*S ;
end

simpson
function int_value=simpson(f,a,b,n)
h=(b-a)/n;
S=0;
x=a;
for i=1:n-1
    x=x+h;
    if mod(i,2)==0
        S=S+2*f(x);
    else
        S=S+4*f(x);
    end
end
int_value=(h/3)*(f(a)+S+f(b));
end

trapezoidal
function int_value = trapezoidal(f,a,b,n)
    h = (b-a)/n;
    S = 0;
    x = a;
    for i = 1:n-1
        x = x + h;
        S = S + f(x);
    end
    int_value = h*(0.5*f(a) + S + 0.5*f(b));
end
