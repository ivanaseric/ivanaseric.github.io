cosTaylor
function [sum, PointError, polynomial] = cosTaylor(x,maxError)
% this function calculates Taylor serieds of erf(x) with error less than maxError
% the error is calculated at each point
    terms_max = 20;
%   first term of the series
    sum = ones(1,length(x));
    n = 1;
    polynomial = zeros(terms_max,1);
    polynomial(1) = 1; % for extra credit
    while (norm(sum - cos(x) ,inf) > maxError)

        sum = sum + ((-1)^n).*x.^(2*n)./factorial(2*n);
        if (n>terms_max) fprintf ('error: Maximum number of terms (%d) reached.\n', terms_max)
            break;
        end
        n = n + 1;
        polynomial(n) = ((-1)^n)/factorial(2*n);    % for extra credit
    end
    PointError = sum - cos(x);
    polynomial = polynomial(1:n); % remove the entries that are zeros

end
